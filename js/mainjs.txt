// LENIS SMOOTH SCROLL
// ============================================
let lenis;

function initLenis() {
  lenis = new Lenis({
    duration: 1.4,
    easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)),
    direction: 'vertical',
    gestureDirection: 'vertical',
    smooth: true,
    smoothTouch: false,
    touchMultiplier: 2,
  });

  function raf(time) {
    lenis.raf(time);
    requestAnimationFrame(raf);
  }
  requestAnimationFrame(raf);

  // Connect Lenis to GSAP ScrollTrigger
  lenis.on('scroll', ScrollTrigger.update);
  gsap.ticker.add((time) => { lenis.raf(time * 1000); });
  gsap.ticker.lagSmoothing(0);

  // Smooth scroll for anchor links
  document.querySelectorAll('a[href^="#"]').forEach(anchor => {
    anchor.addEventListener('click', (e) => {
      const target = document.querySelector(anchor.getAttribute('href'));
      if (target) {
        e.preventDefault();
        lenis.scrollTo(target);
      }
    });
  });
}

// LOADING SCREEN
// ============================================
class Loader {
  constructor() {
    this.screen = document.querySelector('.loading-screen');
    this.bar = document.querySelector('.loading-bar-fill');
    this.percent = document.querySelector('.loading-percent');
    this.progress = 0;
  }

  init() {
    const images = document.querySelectorAll('img');
    let loaded = 0;
    const total = images.length;

    if (total === 0) { this.finish(); return; }

    images.forEach(img => {
      if (img.complete) {
        loaded++;
        this.update(loaded / total);
      } else {
        img.addEventListener('load', () => {
          loaded++;
          this.update(loaded / total);
        });
        img.addEventListener('error', () => {
          loaded++;
          this.update(loaded / total);
        });
      }
    });

    // Fallback timer
    setTimeout(() => this.finish(), 4000);
  }

  update(ratio) {
    const pct = Math.min(Math.round(ratio * 100), 99);
    if (this.bar) this.bar.style.width = pct + '%';
    if (this.percent) this.percent.textContent = pct + '%';
    if (pct >= 99) setTimeout(() => this.finish(), 300);
  }

  finish() {
    if (!this.screen || this.screen.dataset.done) return;
    this.screen.dataset.done = 'true';

    const tl = gsap.timeline();
    tl.to(this.bar, { width: '100%', duration: 0.3, ease: 'power2.out' })
      .to(this.screen, {
        yPercent: -100,
        duration: 1.2,
        ease: 'power4.inOut',
        onComplete: () => {
          this.screen.style.display = 'none';
          document.dispatchEvent(new CustomEvent('siteReady'));
        }
      }, '+=0.2');
  }
}

class Cursor {
  constructor() {
    this.outer = document.querySelector('.cursor-outer');
    this.inner = document.querySelector('.cursor-inner');
    this.pos = { x: window.innerWidth / 2, y: window.innerHeight / 2 };
    this.target = { x: window.innerWidth / 2, y: window.innerHeight / 2 };
    this.isTouch = window.matchMedia('(pointer: coarse)').matches;
    if (this.isTouch) { this.hide(); return; }
    this.init();
  }

  hide() {
    if (this.outer) this.outer.style.display = 'none';
    if (this.inner) this.inner.style.display = 'none';
  }

  init() {
    document.addEventListener('mousemove', (e) => {
      this.target.x = e.clientX;
      this.target.y = e.clientY;
    });

    document.addEventListener('mousedown', () => this.outer?.classList.add('click'));
    document.addEventListener('mouseup', () => this.outer?.classList.remove('click'));

    const interactives = document.querySelectorAll('a, button, .btn, .card-hover, .img-wrap, .gallery-item, .book-showcase, .blog-card');
    interactives.forEach(el => {
      el.addEventListener('mouseenter', () => this.outer?.classList.add('hover'));
      el.addEventListener('mouseleave', () => this.outer?.classList.remove('hover'));
    });

    this.animate();
  }

  animate() {
    this.pos.x += (this.target.x - this.pos.x) * 0.12;
    this.pos.y += (this.target.y - this.pos.y) * 0.12;

    if (this.outer) {
      this.outer.style.left = this.pos.x + 'px';
      this.outer.style.top = this.pos.y + 'px';
    }
    if (this.inner) {
      this.inner.style.left = this.target.x + 'px';
      this.inner.style.top = this.target.y + 'px';
    }

    requestAnimationFrame(() => this.animate());
  }
}

// THREE.JS ATMOSPHERIC BACKGROUND
// ============================================
class Atmosphere {
  constructor(canvasId) {
    this.canvas = document.getElementById(canvasId);
    if (!this.canvas || typeof THREE === 'undefined') return;

    this.scene = new THREE.Scene();
    this.camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
    this.renderer = new THREE.WebGLRenderer({ canvas: this.canvas, alpha: true, antialias: true, powerPreference: 'high-performance' });

    this.time = 0;
    this.mouse = { x: 0, y: 0 };
    this.particles = [];

    this.init();
  }

  init() {
    this.renderer.setSize(window.innerWidth, window.innerHeight);
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    this.renderer.setClearColor(0x08080e, 1);

    this.camera.position.z = 4;

    this.createParticles();
    this.createLightRays();
    this.createDust();
    this.addEvents();
    this.animate();
  }

  createParticles() {
    const count = 200;
    const geometry = new THREE.BufferGeometry();
    const positions = new Float32Array(count * 3);
    const colors = new Float32Array(count * 3);
    const sizes = new Float32Array(count);

    const palette = [
      new THREE.Color(0xd4a574),
      new THREE.Color(0xe8c9a8),
      new THREE.Color(0x5a9a9a),
      new THREE.Color(0xe6e6f0),
      new THREE.Color(0xb88a5a),
    ];

    for (let i = 0; i < count; i++) {
      positions[i * 3] = (Math.random() - 0.5) * 25;
      positions[i * 3 + 1] = (Math.random() - 0.5) * 25;
      positions[i * 3 + 2] = (Math.random() - 0.5) * 12;

      const c = palette[Math.floor(Math.random() * palette.length)];
      colors[i * 3] = c.r;
      colors[i * 3 + 1] = c.g;
      colors[i * 3 + 2] = c.b;

      sizes[i] = Math.random() * 4 + 0.5;
    }

    geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));
    geometry.setAttribute('color', new THREE.BufferAttribute(colors, 3));
    geometry.setAttribute('size', new THREE.BufferAttribute(sizes, 1));

    const material = new THREE.PointsMaterial({
      size: 0.04,
      vertexColors: true,
      transparent: true,
      opacity: 0.55,
      blending: THREE.AdditiveBlending,
      sizeAttenuation: true,
      depthWrite: false,
    });

    this.particleSystem = new THREE.Points(geometry, material);
    this.scene.add(this.particleSystem);
  }

  createLightRays() {
    for (let i = 0; i < 6; i++) {
      const geo = new THREE.PlaneGeometry(18, 0.4, 1, 1);
      const mat = new THREE.MeshBasicMaterial({
        color: 0xd4a574,
        transparent: true,
        opacity: 0.025,
        side: THREE.DoubleSide,
        blending: THREE.AdditiveBlending,
        depthWrite: false,
      });
      const ray = new THREE.Mesh(geo, mat);
      ray.position.set(
        (Math.random() - 0.5) * 12,
        (Math.random() - 0.5) * 12,
        -3 - Math.random() * 4
      );
      ray.rotation.z = Math.random() * Math.PI;
      ray.userData = {
        speed: 0.0008 + Math.random() * 0.0015,
        offset: Math.random() * Math.PI * 2,
        baseOpacity: 0.015 + Math.random() * 0.02,
      };
      this.scene.add(ray);
      this.particles.push(ray);
    }
  }

  createDust() {
    const count = 80;
    const geo = new THREE.BufferGeometry();
    const pos = new Float32Array(count * 3);
    for (let i = 0; i < count; i++) {
      pos[i * 3] = (Math.random() - 0.5) * 20;
      pos[i * 3 + 1] = (Math.random() - 0.5) * 20;
      pos[i * 3 + 2] = (Math.random() - 0.5) * 8;
    }
    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
    const mat = new THREE.PointsMaterial({
      color: 0xd4a574,
      size: 0.015,
      transparent: true,
      opacity: 0.2,
      blending: THREE.AdditiveBlending,
    });
    this.dust = new THREE.Points(geo, mat);
    this.scene.add(this.dust);
  }

  addEvents() {
    window.addEventListener('resize', () => {
      this.camera.aspect = window.innerWidth / window.innerHeight;
      this.camera.updateProjectionMatrix();
      this.renderer.setSize(window.innerWidth, window.innerHeight);
    });

    document.addEventListener('mousemove', (e) => {
      this.mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
      this.mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;
    });
  }

  animate() {
    this.time += 0.008;

    if (this.particleSystem) {
      const pos = this.particleSystem.geometry.attributes.position.array;
      for (let i = 0; i < pos.length; i += 3) {
        pos[i + 1] += Math.sin(this.time * 0.5 + pos[i] * 0.3) * 0.0015;
        pos[i] += Math.cos(this.time * 0.3 + pos[i + 1] * 0.3) * 0.0008;
      }
      this.particleSystem.geometry.attributes.position.needsUpdate = true;
      this.particleSystem.rotation.y = this.time * 0.015;
    }

    this.particles.forEach(ray => {
      const op = ray.userData.baseOpacity + Math.sin(this.time * ray.userData.speed * 120 + ray.userData.offset) * 0.008;
      ray.material.opacity = Math.max(0.005, op);
      ray.rotation.z += ray.userData.speed;
    });

    if (this.dust) {
      this.dust.rotation.y = this.time * 0.008;
      this.dust.rotation.x = this.time * 0.004;
    }

    this.camera.position.x += (this.mouse.x * 0.4 - this.camera.position.x) * 0.04;
    this.camera.position.y += (this.mouse.y * 0.4 - this.camera.position.y) * 0.04;
    this.camera.lookAt(0, 0, 0);

    this.renderer.render(this.scene, this.camera);
    requestAnimationFrame(() => this.animate());
  }
}

// GSAP SCROLL ANIMATIONS
// ============================================
class ScrollAnimations {
  constructor() {
    gsap.registerPlugin(ScrollTrigger);
    this.init();
  }

  init() {
    this.animateElements();
    this.animateImages();
    this.animateParallax();
    this.animateCounters();
    this.animateMarquee();
  }

  animateElements() {
    document.querySelectorAll('[data-animate]').forEach(el => {
      const type = el.dataset.animate || 'up';
      const delay = parseFloat(el.dataset.delay) || 0;
      const duration = parseFloat(el.dataset.duration) || 1;

      let fromVars = { opacity: 0, y: 60 };
      if (type === 'left') fromVars = { opacity: 0, x: -60 };
      if (type === 'right') fromVars = { opacity: 0, x: 60 };
      if (type === 'fade') fromVars = { opacity: 0 };
      if (type === 'scale') fromVars = { opacity: 0, scale: 0.85 };

      gsap.fromTo(el, fromVars, {
        opacity: 1,
        x: 0,
        y: 0,
        scale: 1,
        duration: duration,
        delay: delay,
        ease: 'power3.out',
        scrollTrigger: {
          trigger: el,
          start: 'top 88%',
          toggleActions: 'play none none none',
        }
      });
    });
  }

  animateImages() {
    document.querySelectorAll('.img-reveal').forEach(el => {
      ScrollTrigger.create({
        trigger: el,
        start: 'top 80%',
        onEnter: () => el.classList.add('revealed'),
        once: true,
      });
    });
  }

  animateParallax() {
    document.querySelectorAll('[data-parallax]').forEach(el => {
      const speed = parseFloat(el.dataset.parallax) || 0.3;
      gsap.to(el, {
        y: () => -speed * 100,
        ease: 'none',
        scrollTrigger: {
          trigger: el,
          start: 'top bottom',
          end: 'bottom top',
          scrub: 1.5,
        }
      });
    });
  }

  animateCounters() {
    document.querySelectorAll('.counter').forEach(el => {
      const target = parseInt(el.dataset.target);
      if (!target) return;

      ScrollTrigger.create({
        trigger: el,
        start: 'top 85%',
        onEnter: () => {
          const obj = { val: 0 };
          gsap.to(obj, {
            val: target,
            duration: 2.5,
            ease: 'power2.out',
            onUpdate: () => {
              el.textContent = Math.floor(obj.val).toLocaleString();
            }
          });
        },
        once: true,
      });
    });
  }

  animateMarquee() {
    const marquees = document.querySelectorAll('.marquee');
    marquees.forEach(m => {
      const track = m.querySelector('.marquee-track');
      if (!track) return;
      const clone = track.cloneNode(true);
      m.appendChild(clone);
    });
  }
}

// NAVIGATION
// ============================================
class Navigation {
  constructor() {
    this.nav = document.querySelector('.main-nav');
    this.toggle = document.querySelector('.menu-toggle');
    this.mobile = document.querySelector('.mobile-menu');
    this.lastScroll = 0;
    this.init();
  }

  init() {
    window.addEventListener('scroll', () => this.onScroll(), { passive: true });

    if (this.toggle) {
      this.toggle.addEventListener('click', () => this.toggleMenu());
    }

    // Close mobile menu on link click
    if (this.mobile) {
      this.mobile.querySelectorAll('a').forEach(a => {
        a.addEventListener('click', () => this.closeMenu());
      });
    }

    // Highlight active nav link
    this.highlightActive();
  }

  onScroll() {
    const y = window.scrollY || window.pageYOffset;
    if (y > 80) {
      this.nav?.classList.add('scrolled');
    } else {
      this.nav?.classList.remove('scrolled');
    }
    this.lastScroll = y;
  }

  toggleMenu() {
    this.toggle?.classList.toggle('active');
    this.mobile?.classList.toggle('active');
    document.body.style.overflow = this.mobile?.classList.contains('active') ? 'hidden' : '';
    if (lenis) {
      this.mobile?.classList.contains('active') ? lenis.stop() : lenis.start();
    }
  }

  closeMenu() {
    this.toggle?.classList.remove('active');
    this.mobile?.classList.remove('active');
    document.body.style.overflow = '';
    if (lenis) lenis.start();
  }

  highlightActive() {
    const path = window.location.pathname;
    document.querySelectorAll('.nav-links a, .mobile-menu a').forEach(link => {
      const href = link.getAttribute('href');
      if (href && path.includes(href.replace('./', '').replace('../', ''))) {
        link.classList.add('active');
      }
    });
  }
}

class MagneticButtons {
  constructor() {
    this.buttons = document.querySelectorAll('.magnetic');
    this.isTouch = window.matchMedia('(pointer: coarse)').matches;
    if (!this.isTouch) this.init();
  }

  init() {
    this.buttons.forEach(btn => {
      btn.addEventListener('mousemove', (e) => this.move(e, btn));
      btn.addEventListener('mouseleave', (e) => this.reset(e, btn));
    });
  }

  move(e, btn) {
    const rect = btn.getBoundingClientRect();
    const x = e.clientX - rect.left - rect.width / 2;
    const y = e.clientY - rect.top - rect.height / 2;
    gsap.to(btn, {
      x: x * 0.25,
      y: y * 0.25,
      duration: 0.4,
      ease: 'power2.out',
    });
  }

  reset(e, btn) {
    gsap.to(btn, {
      x: 0,
      y: 0,
      duration: 0.7,
      ease: 'elastic.out(1, 0.35)',
    });
  }
}

// 3D CARD TILT
// ============================================
class CardTilt {
  constructor() {
    this.cards = document.querySelectorAll('.tilt-card');
    this.isTouch = window.matchMedia('(pointer: coarse)').matches;
    if (!this.isTouch) this.init();
  }

  init() {
    this.cards.forEach(card => {
      card.addEventListener('mousemove', (e) => this.tilt(e, card));
      card.addEventListener('mouseleave', (e) => this.reset(e, card));
    });
  }

  tilt(e, card) {
    const rect = card.getBoundingClientRect();
    const x = (e.clientX - rect.left) / rect.width;
    const y = (e.clientY - rect.top) / rect.height;
    const rx = (y - 0.5) * -12;
    const ry = (x - 0.5) * 12;

    gsap.to(card, {
      rotateX: rx,
      rotateY: ry,
      duration: 0.5,
      ease: 'power2.out',
      transformPerspective: 1200,
    });
  }

  reset(e, card) {
    gsap.to(card, {
      rotateX: 0,
      rotateY: 0,
      duration: 0.8,
      ease: 'elastic.out(1, 0.4)',
    });
  }
}

class NewsletterForm {
  constructor() {
    this.forms = document.querySelectorAll('.newsletter-form');
    this.init();
  }

  init() {
    this.forms.forEach(form => {
      form.addEventListener('submit', (e) => this.handle(e, form));
    });
  }

  handle(e, form) {
    e.preventDefault();
    const email = form.querySelector('input[type="email"]')?.value;
    if (!email) return;

    const btn = form.querySelector('button');
    const original = btn.innerHTML;

    gsap.to(btn, {
      scale: 0.95,
      duration: 0.1,
      yoyo: true,
      repeat: 1,
    });

    const success = document.createElement('div');
    success.className = 'form-success';
    success.innerHTML = `
      <div style="font-family:var(--font-display);font-size:1.3rem;color:var(--rose-gold);margin-bottom:0.5rem;">Welcome to the story.</div>
      <div style="font-size:0.9rem;color:var(--warm-gray);">Check your inbox for confirmation.</div>
    `;
    success.style.cssText = 'position:absolute;inset:0;display:flex;flex-direction:column;justify-content:center;align-items:center;background:var(--charcoal);border-radius:3px;opacity:0;z-index:10;';

    form.style.position = 'relative';
    form.appendChild(success);

    gsap.to(success, { opacity: 1, duration: 0.6, ease: 'power2.out' });

    setTimeout(() => {
      gsap.to(success, {
        opacity: 0,
        duration: 0.5,
        onComplete: () => success.remove(),
      });
      form.reset();
    }, 3500);
  }
}

// SPLIT TEXT ANIMATION
// ============================================
class SplitTextAnim {
  constructor() {
    document.querySelectorAll('.split-text').forEach(el => this.split(el));
  }

  split(el) {
    const text = el.textContent;
    el.innerHTML = '';
    el.style.overflow = 'hidden';
    el.style.display = 'inline-block';

    const wrapper = document.createElement('span');
    wrapper.style.display = 'inline-block';

    text.split('').forEach((char, i) => {
      const span = document.createElement('span');
      span.style.display = 'inline-block';
      span.style.opacity = '0';
      span.style.transform = 'translateY(100%)';
      span.textContent = char === ' ' ? '\u00A0' : char;
      wrapper.appendChild(span);
    });

    el.appendChild(wrapper);

    ScrollTrigger.create({
      trigger: el,
      start: 'top 85%',
      onEnter: () => {
        gsap.to(wrapper.children, {
          opacity: 1,
          y: 0,
          duration: 0.7,
          stagger: 0.03,
          ease: 'power3.out',
        });
      },
      once: true,
    });
  }
}

// QUOTE ROTATOR
// ============================================
class QuoteRotator {
  constructor() {
    this.container = document.querySelector('.quote-rotator');
    if (!this.container) return;
    this.quotes = this.container.querySelectorAll('.quote-item');
    this.current = 0;
    this.init();
  }

  init() {
    if (this.quotes.length <= 1) return;
    this.quotes.forEach((q, i) => {
      if (i !== 0) { q.style.opacity = '0'; q.style.position = 'absolute'; }
    });
    setInterval(() => this.rotate(), 7000);
  }

  rotate() {
    const curr = this.quotes[this.current];
    const next = this.quotes[(this.current + 1) % this.quotes.length];

    gsap.to(curr, {
      opacity: 0,
      y: -25,
      duration: 0.7,
      ease: 'power2.in',
      onComplete: () => {
        curr.style.position = 'absolute';
        next.style.position = 'relative';
        gsap.fromTo(next,
          { opacity: 0, y: 25 },
          { opacity: 1, y: 0, duration: 1, ease: 'power3.out' }
        );
      }
    });

    this.current = (this.current + 1) % this.quotes.length;
  }
}

class Lightbox {
  constructor() {
    this.items = document.querySelectorAll('[data-lightbox]');
    this.overlay = null;
    this.init();
  }

  init() {
    this.items.forEach(item => {
      item.addEventListener('click', (e) => this.open(e, item));
    });
  }

  open(e, item) {
    e.preventDefault();
    const src = item.querySelector('img')?.src || item.dataset.lightbox;
    if (!src) return;

    this.overlay = document.createElement('div');
    this.overlay.style.cssText = `
      position:fixed;inset:0;background:rgba(8,8,14,0.95);z-index:9999;
      display:flex;justify-content:center;align-items:center;
      opacity:0;cursor:pointer;
    `;

    const img = document.createElement('img');
    img.src = src;
    img.style.cssText = 'max-width:90vw;max-height:85vh;object-fit:contain;border-radius:4px;box-shadow:0 40px 80px rgba(0,0,0,0.5);';

    this.overlay.appendChild(img);
    document.body.appendChild(this.overlay);

    gsap.to(this.overlay, { opacity: 1, duration: 0.5, ease: 'power2.out' });
    gsap.fromTo(img, { scale: 0.9, opacity: 0 }, { scale: 1, opacity: 1, duration: 0.6, ease: 'power3.out', delay: 0.1 });

    this.overlay.addEventListener('click', () => this.close());
    if (lenis) lenis.stop();
  }

  close() {
    if (!this.overlay) return;
    gsap.to(this.overlay, {
      opacity: 0,
      duration: 0.4,
      onComplete: () => {
        this.overlay.remove();
        this.overlay = null;
        if (lenis) lenis.start();
      }
    });
  }
}
